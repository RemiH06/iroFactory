Código

Inline: toggleTheme() cambia la carta de día a carta estelar.

# Bitácora del viajero
class Destino:
    def __init__(self, nombre, lat, lon):
        self.nombre   = nombre
        self.coords   = (lat, lon)
        self.visitado = False

    def marcar(self, nota=None):
        self.visitado = True
        return f"{self.nombre} · registrado"

ruta = [
    Destino("Roma",  41.89, 12.49),
    Destino("Petra", 30.32, 35.45),
