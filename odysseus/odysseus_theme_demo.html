# Bitácora del viajero
class Destino:
    def __init__(self, nombre, lat, lon):
        self.nombre   = nombre
        self.coords   = (lat, lon)
        self.visitado = False
        self.notas    = []

    def marcar(self, nota=None):
        self.visitado = True
        if nota: self.notas.append(nota)
        return f"✓ {self.nombre} — registrado"
