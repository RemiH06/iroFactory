# El oráculo de Delfos
class Oraculo:
    def __init__(self, dios):
        self.dios    = dios
        self.activo  = True

    def consultar(self, pregunta):
        return f"Apolo habla: {pregunta[::-1]}"

apolo = Oraculo("Apolo")
print(apolo.consultar("¿Cuál es mi destino?"))
