# Índice de un bosque de libros
class Libro:
    def __init__(self, titulo, autor, paginas):
        self.titulo  = titulo
        self.autor   = autor
        self.paginas = paginas
        self.leido   = False

    def leer(self):
        self.leido = True
        return f"'{self.titulo}' · terminado"
