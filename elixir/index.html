# Pipeline de clasificación supervisada
from sklearn.pipeline import Pipeline
from sklearn.preprocessing import LabelEncoder

@dataclass
class Experimento:
    nombre: str
    epochs: int  = 50
    lr:     float = 1e-4

    def entrenar(self, X, y):
        modelo = Pipeline([
            ('enc', LabelEncoder()),
            ('clf', self._build()),
        ])
        return modelo.fit(X, y)   # clasificador listo
