Código

Inline: toggleTheme() alterna entre carpa y feria de día.

-- Programación de actos de la noche
local ACTOS = { "malabaristas", "equilibristas", "cañón humano" }

function programarNoche(actos, pista)
  for i, acto in ipairs(actos) do
    print("Pista " .. pista .. ": " .. acto)
  end
end

-- La carpa principal tiene 3 pistas simultáneas
programarNoche(ACTOS, 1)
