-- Lógica de puntuación de Tetris
local SCORES = { 100, 300, 500, 800 }

function calcularPuntos(lineas, nivel)
  local base = SCORES[lineas] or 0
  return base * (nivel + 1)
end

-- Tetris (4 líneas) en nivel 18 = 800 * 19 = 15,200
print(calcularPuntos(4, 18))
